# Replay the real loop under a continuously queued input and long AFE
# preemption. Recovery must follow wall time, including failed inferences.
add_executable(kws_stream_slow_runtime_test stream_runtime_test.cpp)
target_compile_features(kws_stream_slow_runtime_test PRIVATE cxx_std_17)
target_include_directories(kws_stream_slow_runtime_test PRIVATE ".." "../include" "../model" "../.." "../../usb_driver" "stubs")
target_compile_definitions(kws_stream_slow_runtime_test PRIVATE
    KWS_TEST_PROCESS_US=35000 CONFIG_KWS_INPUT_USB_PLAYBACK_REFERENCE=1)
if(MSVC)
    target_compile_options(kws_stream_slow_runtime_test PRIVATE /UNDEBUG)
else()
    target_compile_options(kws_stream_slow_runtime_test PRIVATE -UNDEBUG)
endif()
add_test(NAME kws_stream_slow_runtime COMMAND kws_stream_slow_runtime_test)
