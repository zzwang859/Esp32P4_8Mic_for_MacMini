cmake_minimum_required(VERSION 3.16)
project(array_dsp_tests LANGUAGES C CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include("../gain_calibration.cmake")
generate_array_gain_header("${Python3_EXECUTABLE}" "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(array_dsp_test array_dsp_test.cpp ../array_dsp.cpp
    reference/doa_algorithms.cpp)
target_include_directories(array_dsp_test PRIVATE .. reference
    "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(array_dsp_test PRIVATE cxx_std_17)
add_test(NAME array_dsp COMMAND array_dsp_test)
add_executable(doa_tracking_test doa_tracking_test.cpp)
target_include_directories(doa_tracking_test PRIVATE ..)
target_compile_features(doa_tracking_test PRIVATE cxx_std_17)
add_test(NAME doa_tracking COMMAND doa_tracking_test)
add_library(nscore_test_core OBJECT ../nscore/enc_core.c ../nscore/ns_fft.c)
add_executable(nscore_test nscore_test.cpp)
target_include_directories(nscore_test PRIVATE ..)
target_compile_features(nscore_test PRIVATE cxx_std_17)
target_link_libraries(nscore_test PRIVATE nscore_test_core)
add_test(NAME nscore COMMAND nscore_test)
