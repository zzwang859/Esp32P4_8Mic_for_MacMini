cmake_minimum_required(VERSION 3.16)
project(custom_control_tests LANGUAGES C)
enable_testing()
add_executable(custom_protocol_test protocol_test.c ../custom_protocol.c)
target_include_directories(custom_protocol_test PRIVATE ..)
add_test(NAME custom_protocol COMMAND custom_protocol_test)
add_executable(custom_state_test state_test.c ../custom_state.c ../custom_protocol.c)
target_include_directories(custom_state_test PRIVATE .. stubs ../../usb_driver)
add_test(NAME custom_state COMMAND custom_state_test)
if(WIN32 AND NOT MSVC)
    set(cjson_dir "${CMAKE_CURRENT_LIST_DIR}/../../../managed_components/espressif__cjson/cJSON")
    if(EXISTS "${cjson_dir}/cJSON.c")
        add_executable(custom_storage_test storage_test.c ../custom_storage.c ../custom_state.c
            ../custom_protocol.c "${cjson_dir}/cJSON.c")
        target_include_directories(custom_storage_test PRIVATE .. stubs ../../usb_driver "${cjson_dir}")
        target_link_libraries(custom_storage_test PRIVATE bcrypt)
        target_compile_options(custom_storage_test PRIVATE -UNDEBUG)
        add_test(NAME custom_storage COMMAND custom_storage_test)
    endif()
endif()
foreach(target custom_protocol_test custom_state_test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
endforeach()
